//Rest parameters: using spread parameters in function parameter
function sum_rest(x, y, ...numbers) {
	let sum = 0;
	for (let number of numbers) {
		sum += number;
	}
	return x + y + sum;
}

function shuffle(array) {
	let cindex = array.length;
	while (cindex != 0) {
		let rindex = Math.floor(Math.random() * array.length);
		cindex -= 1;

		let temp = array[cindex];
		array[cindex] = array[rindex];
		array[rindex] = temp;
	}
	return array;
}

class Animal {
	alive = true;
	name = "Animal";

	constructor(name, status) {
		this.name = name;
		this.alive = status;
	}

	eat() {
		console.log(this.name, "eats");
	}
	sleeps() {
		console.log(this.name, "sleeps");
	}
}
class Rabbit extends Animal {
	name = "Rabbit";

	constructor(name, status, runSpeed) {
		super(name, status);
		this.runSpeed = runSpeed;
	}
	run() {
		console.log(this.name, "runs");
	}
}
class Hawk extends Animal {
	name = "Hawk";

	constructor(name, status, flySpeed) {
		super(name, status);
		this.flySpeed = flySpeed;
	}
	fly() {
		console.log(this.name, "flies");
	}
}

class Fish extends Animal {
	name = "Fish";

	constructor(name, status, swimSpeed) {
		super(name, status);
		this.swimSpeed = swimSpeed;
		this._type = name;
	}
	swim() {
		console.log(this.name, "swims");
	}
	//_attribute: used to write a protected attribute in a class for a getter method,
	//Get:

		console.log("Trout");
	}
	//set(): binds a value to a function
	set type(value) {
		//i.e. doesn't matter if value is trout or not it will be set to Trout
		console.log("Trout");
		this._type = "Trout";
	}
}
