functions: 
     selectors:
          document.firstElementChild, lastElementChild //select by position
          
          querySelector("") //returns the first Element within the document that matches the specified selector,
               //useful for tagname, #id, .class
               //Combining selectors: querySelector("") is same as CSS rules in quotes
          querySelectorAll() //list of all elements matching the specified selectors
          
          getElementbyId //select by id

          getElementByTagName //collection of objects with same tag name i.e. object name
          getElementByTagName(element)[no. in arr].style //for a particular object in array

          getElementbyClassName(element)[0] //even if class only has one element [0] is added

          //bootstraping by javascript
          (element).classList.add/remove/toggle ("class") //and then you can  just add the properties to the class using stylesheet


     manipulators:

          //manipulating text:
          (element).innerHTML //gives the innerHTML of the element, useful to change the HTML using javascript
          (element).textContent //change the text content inside the HTML element

          //manipulating attributes:
          (element).getAttribute("attribute") //get a single attribute, when getAttribute("attribute") is not given answer is null 
          (element).setAttribute("attribute", "value") //set the attribute

          (element).style.colour //to change style options; here colour
          (element).click //performs click action on element
          
          (element).onClick //useful for buttons; after action

Functions as arguments: i.e. Higher order functions
     example: //notice how operator does not require ();
     function calculator(a, b, operator){

     }
     function add(a, b){
          return a+b;
     }
     function multiply(a, b){
          return a*b;
